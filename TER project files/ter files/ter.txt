    private static Map<String, List<String[]>> parseSAKeyOutput(String saKeyOutput) {
        // Initialize a map to hold the results
        Map<String, List<String[]>> keys = new HashMap<>();
        keys.put("1-non", new ArrayList<>());
        keys.put("0-almost", new ArrayList<>());
        
        // Define regular expressions to match the keys
        String nonKeysRegex = "1-non keys:\\s*\\[\\[([^]]+)\\]";
        String almostKeysRegex = "0-almost keys:\\s*\\[\\[([^]]+)\\]";

        // Use regular expressions to find all matches for non and almost keys
        Pattern nonKeysPattern = Pattern.compile(nonKeysRegex);
        Pattern almostKeysPattern = Pattern.compile(almostKeysRegex);

        // Match non-keys
        Matcher nonKeysMatcher = nonKeysPattern.matcher(saKeyOutput);
        if (nonKeysMatcher.find()) {
            String nonKeysData = nonKeysMatcher.group(1);
            processKeyData(nonKeysData, keys.get("1-non"));
        }

        // Match almost-keys
        Matcher almostKeysMatcher = almostKeysPattern.matcher(saKeyOutput);
        if (almostKeysMatcher.find()) {
            String almostKeysData = almostKeysMatcher.group(1);
            processKeyData(almostKeysData, keys.get("0-almost"));
        }

        return keys;
    }

    private static void processKeyData(String keyData, List<String[]> keyList) {
        // Split the keys by "], ["
        String[] keyGroups = keyData.split("\\],\\s*\\[");

        // Process each key group
        for (String keyGroup : keyGroups) {
            // Clean the key group and split by comma
            String[] keys = keyGroup.replace("[", "").replace("]", "").split(",\\s*");
            keyList.add(keys);
        }
    }
