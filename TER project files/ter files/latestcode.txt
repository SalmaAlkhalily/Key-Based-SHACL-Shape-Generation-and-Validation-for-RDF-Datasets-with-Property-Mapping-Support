    private File processDataFile3(File inputFile) {
        if (inputFile == null || !inputFile.exists()) {
            System.err.println("Input file does not exist.");
            return null;
        }

        File processedFile = new File("processed_data.ttl");
        String namespace = "http://example.com/ns#";
        System.out.println("Processing input file...");

        try (BufferedReader reader = new BufferedReader(new FileReader(inputFile));
             BufferedWriter writer = new BufferedWriter(new FileWriter(processedFile))) {

            String line;
            long lineCount = 0;

            while ((line = reader.readLine()) != null) {
                String[] parts = line.split("\\s+");
                if (parts.length >= 3) {
                    writer.write(String.format("<%s%s> <%s%s> %s .%n",
                            namespace, parts[0].replace("<", "").replace(">", ""),
                            namespace, parts[1].replace("<", "").replace(">", "").toLowerCase(),
                            parts[2]));
                    lineCount++;

                    if (lineCount % 1000 == 0) {
                        System.out.println("Processed " + lineCount + " lines...");
                    }
                }
            }

            System.out.println("Finished processing " + lineCount + " lines.");

        } catch (IOException e) {
            System.err.println("Error processing data file: " + e.getMessage());
            return null;
        }

        return processedFile;
    }

    private File processDataFile(String inputFilePath) {
        System.setProperty("hadoop.native.io.disabled", "true");
        SparkSession spark = SparkSession.builder()
                .appName("RDF Processor")
                .master("local[*]") // Use "yarn" for cluster deployment
                .getOrCreate();

        // Input and output file paths
        String outputPath = "output/";

        // Prefix and inverse property suffix
        String prefix = "http://example.com/ns#";
        String inversePropertySuffix = "inv";

        // Read input RDF file as a Dataset
        Dataset<Row> rdfData = spark.read().text(inputFilePath);

        // Process the RDF triples
        Dataset<Row> transformedData = rdfData
                .filter((FilterFunction<Row>) row -> !row.getString(0).trim().isEmpty()) // Filter out empty lines
                .map((MapFunction<Row, String>) row -> {
                    String line = row.getString(0).trim();
                    String[] parts = line.split(" ", 3);
                    if (parts.length != 3) return ""; // Skip invalid lines

                    // Extract subject, predicate, and object
                    String subject = parts[0].replace("<", "").replace(">", "");
                    String predicate = parts[1].replace("<", "").replace(">", "");
                    String object = parts[2].replace("<", "").replace(">", "").replace(".", "");
                    object = object.replace("\"", "");


                    // Transform components
                    String transformedSubject = prefix + subject.replace("`", "_");
                    String transformedPredicate = prefix + predicate.toLowerCase().replace("`", "_") + inversePropertySuffix;
                    String transformedObject = "\"" + object.replace("`", "_") + "\"";

                    if (transformedObject.startsWith("\"")
                            && transformedObject.contains("-")
                            && transformedObject.contains("##")) {
                        transformedObject = transformedObject
                                .replace("\"","")
                                .replace("##","01");
                        transformedObject = "\"" + transformedObject + "\"";
                    }

                    // Return the transformed triple
                    return "<" + transformedSubject + "> <" + transformedPredicate + "> " + transformedObject + " .";
                }, Encoders.STRING()).toDF();

        // Coalesce the data to a single partition to ensure one output file
        transformedData.coalesce(1).write().mode(SaveMode.Overwrite).text(outputPath);

        System.out.println("Processing complete. Output written to " + outputPath);

        // Stop Spark session
        spark.stop();

        // List the files in the output directory
        File outputDir = new File(outputPath);
        File[] files = outputDir.listFiles((dir, name) -> name.startsWith("part-"));

        if (files != null && files.length > 0) {
            // Get the first matching file (it should be the only one if coalesce(1) was used)
            File partFile = files[0];

            // Define the target path for renaming
            Path targetPath = Paths.get(outputPath + "processed_data.ttl");

            try {
                // Rename the part file to the desired name
                Files.move(partFile.toPath(), targetPath);
                return targetPath.toFile();
            } catch (Exception e) {
                e.printStackTrace();
                throw new RuntimeException("Failed to rename the output file.", e);
            }
        } else {
            throw new RuntimeException("No part files found in the output directory.");
        }
    }



    private File processing(String inputFilePath, String outputFilePath, String namespace) throws IOException {
        // Set up Spark configuration
        SparkConf conf = new SparkConf().setAppName("FileProcessor");
        // For local testing, set master to "local[*]" to use all cores
        // For cluster deployment, remove this line or set it appropriately
        conf.setMaster("local[*]");

        try (JavaSparkContext sc = new JavaSparkContext(conf)) {

            // Read the input file as an RDD (Resilient Distributed Dataset)
            JavaRDD<String> lines = sc.textFile(inputFilePath);

            // Process each line using a map transformation
            JavaRDD<String> processedLines = lines.map(new Function<String, String>() {
                @Override
                public String call(String line) throws Exception {
                    String[] parts = line.split("\\s+");
                    if (parts.length >= 3) {
                        String subject = parts[0].replace("<", "").replace(">", "");
                        String predicate = parts[1].replace("<", "").replace(">", "").toLowerCase();
                        return String.format("<%s%s> <%s%s> %s .", namespace, subject, namespace, predicate, parts[2]);
                    } else {
                        return ""; // Or handle invalid lines differently (e.g., log them)
                    }
                }
            }).filter(s -> !s.isEmpty()); // Remove empty lines

            // Save the processed data to a new file
            processedLines.saveAsTextFile(outputFilePath);

            // Optionally, collect the results to a local list (use with caution for large datasets)
            // List<String> outputLines = processedLines.collect();
            // Files.write(Paths.get("local_output.txt"), outputLines, StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING);

        }
        return new File(outputFilePath);
    }

    private void generateShapes(List<List<String>> almostKeys) {
        shapesTextArea.clear(); // Clear previous content
        saveShapesButton.setDisable(true); // Disable the save button during generation

        if (almostKeys == null || almostKeys.isEmpty()) {
            showAlert("Error", "No almost keys available to generate shapes.", Alert.AlertType.ERROR);
            return;
        }

        Model model = ModelFactory.createDefaultModel();

        // Define prefixes
        String sh = "http://www.w3.org/ns/shacl#";
        String ex = "http://example.com/ns#";

        model.setNsPrefix("sh", sh);
        model.setNsPrefix("ex", ex);
        model.setNsPrefix("xsd", XSD.getURI());

        Property rdfType = RDF.type;

        Resource sparqlTarget = model.createResource(sh + "SPARQLTarget");
        Property shTarget = model.createProperty(sh + "target");
        Property shPath = model.createProperty(sh + "path");
        Property shProperty = model.createProperty(sh + "property");
        Property shDatatype = model.createProperty(sh + "datatype");
        Property shMaxCount = model.createProperty(sh + "maxCount");
        Property shSPARQLConstraint = model.createProperty(sh + "SPARQLConstraint");
        Property shSelect = model.createProperty(sh + "select");
        Property shMessage = model.createProperty(sh + "message");
        Property shResultPath = model.createProperty(sh + "resultPath");
        Property shSeverity = model.createProperty(sh + "severity");

        Resource shViolation = model.createResource(sh + "Violation");

        int shapeIndex = 1;
        for (List<String> key : almostKeys) {
            Resource shape = model.createResource(ex + "Shape" + shapeIndex++);
            shape.addProperty(rdfType, model.createResource(sh + "NodeShape"));

            // Add target definition
            Resource target = model.createResource();
            target.addProperty(rdfType, sparqlTarget);
            target.addProperty(shSelect, """
                        PREFIX ex: <http://example.com/ns#>
                        SELECT ?node
                        WHERE {
                            ?node ex:%s ?value .
                        }
                    """.formatted(key.get(0))); // Adjust for the first property in the list
            shape.addProperty(shTarget, target);

            // Add properties
            for (String property : key) {
                Resource propertyShape = model.createResource();
                propertyShape.addProperty(shPath, model.createProperty(ex + property));
                propertyShape.addProperty(shDatatype, XSD.xstring);
                propertyShape.addProperty(shMaxCount, model.createTypedLiteral(1));
                shape.addProperty(shProperty, propertyShape);
            }

            // Add SPARQL constraint for duplicate detection
            Resource sparqlConstraint = model.createResource();
            sparqlConstraint.addProperty(RDF.type, model.createResource(sh + "SPARQLConstraint")); // Important: Still of type sh:SPARQLConstraint
            sparqlConstraint.addProperty(shMessage, "Duplicate detected for properties: " + String.join(", ", key));

            String sparqlSelectQuery = String.format("""
                                PREFIX ex: <http://example.com/ns#>
                                SELECT ?this
                                WHERE {
                                  ?this ex:%s ?value .
                                  ?other ex:%s ?value .
                                  FILTER (?this != ?other)
                                }
                            """,
                    key.get(0),
                    key.get(0)
            );

            sparqlConstraint.addProperty(shSelect, sparqlSelectQuery);
            // Remove shResultPath. It's not standard with sh:sparql in this context.
            // sparqlConstraint.addProperty(shResultPath, model.createProperty(ex + key.get(0)));
            shape.addProperty(model.createProperty(sh + "sparql"), sparqlConstraint); // Use sh:sparql here
            shape.addProperty(shSeverity, shViolation);
        }

        // Serialize the model to Turtle format and display in the shapesTextArea
        StringWriter writer = new StringWriter();
        model.write(writer, "TURTLE");
        String generatedShapes = writer.toString();

        // Update UI on the JavaFX thread
        javafx.application.Platform.runLater(() -> {
            shapesTextArea.replaceText(generatedShapes); // Display shapes in the text area
            saveShapesButton.setDisable(false); // Enable save button
        });
    }

    private void openRdfFileChooser() {
        FileChooser fileChooser = new FileChooser();
        fileChooser.getExtensionFilters().add(new FileChooser.ExtensionFilter("RDF Files", "*.nt", "*.ttl"));
        File selectedFile = fileChooser.showOpenDialog(null);
        if (selectedFile != null) {
            selectedFilePath = selectedFile.getAbsolutePath();
            rdfContentTextArea.clear();
            almostKeys = new ArrayList<>();
            outputTextArea.clear();
            saveKeysButton.setDisable(true);
            loadingIndicator.setVisible(true);
            rdfContentTextArea.replaceText("Loading file content... Please wait.");
            new Thread(() -> {
                try {
                    String content = new String(Files.readAllBytes(Paths.get(selectedFile.toURI())));
                    javafx.application.Platform.runLater(() -> {
                        rdfContentTextArea.replaceText(content);
                        loadingIndicator.setVisible(false);
                        generateKeysButton.setDisable(false);
                        generateKeysButton.setOnAction(e -> generateKeys());
                    });
                } catch (IOException ex) {
                    ex.printStackTrace();
                    javafx.application.Platform.runLater(() -> {
                        showAlert("Error", "Failed to read the file.", Alert.AlertType.ERROR);
                        loadingIndicator.setVisible(false);
                    });
                }
            }).start();
        }
    }

    private void generateKeys() {
        if (selectedFilePath != null && !selectedFilePath.isEmpty()) {
            String sakeyCommand = "java -jar C:\\Users\\asus\\Downloads\\TER-Materials\\TER-Materials\\sakey.jar \"" + selectedFilePath + "\" 1";
            loadingIndicator.setVisible(true);
            outputTextArea.clear();
            new Thread(() -> {
                try {
                    Process process = new ProcessBuilder(sakeyCommand.split(" ")).start();
                    BufferedReader reader = new BufferedReader(new InputStreamReader(process.getInputStream()));
                    StringBuilder output = new StringBuilder();
                    String line;
                    while ((line = reader.readLine()) != null) {
                        output.append(line).append("\n");
                    }

                    int exitCode = process.waitFor();
                    javafx.application.Platform.runLater(() -> {
                        loadingIndicator.setVisible(false);
                        if (exitCode == 0) {
                            outputTextArea.setText(output.toString());
                            almostKeys = parseAlmostKeysFromOutput(output.toString());
                            saveKeysButton.setDisable(false);
                        } else {
                            showAlert("Error", "The sakey tool encountered an error.", Alert.AlertType.ERROR);
                        }
                    });
                } catch (IOException | InterruptedException ex) {
                    ex.printStackTrace();
                    javafx.application.Platform.runLater(() -> {
                        showAlert("Error", "An error occurred while running the sakey tool.", Alert.AlertType.ERROR);
                        loadingIndicator.setVisible(false);  // Hide the spinner on error
                    });
                }
            }).start();
        } else {
            showAlert("Error", "No RDF file selected.", Alert.AlertType.ERROR);
        }
    }

    private List<List<String>> parseAlmostKeysFromOutput(String output) {
        List<List<String>> keys = new ArrayList<>();
        String almostKeysOutput = output.split("0-almost keys:")[1].split("1-non keys:")[0].trim();
        String[] keyGroups = almostKeysOutput.split("\\], \\[");
        for (String group : keyGroups) {
            List<String> keyList = Arrays.asList(group.replace("[", "").replace("]", "").split(", "));
            keys.add(keyList);
        }

        return keys;
    }

    private void showAlert(String title, String message, Alert.AlertType alertType) {
        Alert alert = new Alert(alertType);
        alert.setTitle(title);
        alert.setHeaderText(null);
        alert.setContentText(message);
        alert.showAndWait();
    }
