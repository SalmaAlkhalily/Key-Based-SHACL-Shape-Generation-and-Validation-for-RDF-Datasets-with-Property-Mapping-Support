    private File processDataFile(String inputFilePath) {
        System.setProperty("hadoop.native.io.disabled", "true");
        SparkSession spark = SparkSession.builder()
                .appName("RDF Processor")
                .master("local[*]")
                .getOrCreate();

        String outputPath = "output/";
        String prefix = "http://example.com/ns#";
        Dataset<Row> rdfData = spark.read().text(inputFilePath);
        Dataset<Row> transformedData = rdfData
                .filter((FilterFunction<Row>) row -> !row.getString(0).trim().isEmpty())
                .map((MapFunction<Row, String>) row -> {
                    String line = row.getString(0).trim();
                    String[] parts = line.split(" ", 3);
                    if (parts.length != 3) return "";
                    String subject = parts[0].replace("<", "").replace(">", "");
                    String predicate = parts[1].replace("<", "").replace(">", "");
                    String object = parts[2].replace("<", "").replace(">", "").replace(".", "");
                    object = object.replace("\"", "");
                    String transformedSubject = prefix + subject.replace("`", "_");
                    String transformedPredicate = prefix + predicate.toLowerCase().replace("`", "_");
                    String transformedObject = "\"" + object.replace("`", "_") + "\"";

                    if (transformedObject.startsWith("\"")
                            && transformedObject.contains("-")
                            && transformedObject.contains("##")) {
                        transformedObject = transformedObject
                                .replace("\"", "")
                                .replace("##", "01");
                        transformedObject = "\"" + transformedObject + "\"";
                    }

                    return "<" + transformedSubject + "> <" + transformedPredicate + "> " + transformedObject + " .";
                }, Encoders.STRING()).toDF();

        transformedData.coalesce(1).write().mode(SaveMode.Overwrite).text(outputPath);

        System.out.println("Processing complete. Output written to " + outputPath);

        spark.stop();

        File outputDir = new File(outputPath);
        File[] files = outputDir.listFiles((dir, name) -> name.startsWith("part-"));

        if (files != null && files.length > 0) {
            File partFile = files[0];
            Path targetPath = Paths.get(outputPath + "processed_data.ttl");

            try {
                Files.move(partFile.toPath(), targetPath);
                return targetPath.toFile();
            } catch (Exception e) {
                throw new RuntimeException("Failed to rename the output file.", e);
            }
        } else {
            throw new RuntimeException("No part files found in the output directory.");
        }
    }


    private void validateRdfData() {
        if (dataFilePath == null || shapesFilePath == null) {
            showAlert("Error", "Please choose both a data file and a shapes file.", Alert.AlertType.ERROR);
            return;
        }

        File processedDataFile;
        processedDataFile = processDataFile(dataFilePath);
        String DATA = processedDataFile.getAbsolutePath();
        String SHAPES = shapesFilePath;
        Graph shapesGraph = GraphFactory.createDefaultGraph();
        RDFDataMgr.read(shapesGraph, SHAPES);

        System.out.println("Data:" + DATA);
        Graph dataGraph = GraphFactory.createDefaultGraph();
        RDFDataMgr.read(dataGraph, DATA);

        Shapes shapes = Shapes.parse(shapesGraph);
        System.out.println("Shapes: " + shapes.getGraph());
        ValidationReport report = ShaclValidator.get().validate(shapes.getGraph(), dataGraph);
        ShLib.printReport(report);
        System.out.println("Report:" + report.getEntries());
        RDFDataMgr.write(System.out, report.getModel(), Lang.NT);
        Map<String, Integer> violationCounts = new HashMap<>();
        AtomicInteger totalViolations = new AtomicInteger(0);
        Model reportModel = report.getModel();
        report.getEntries().forEach(entry -> {
            Node sourceConstraintNode = entry.sourceConstraintComponent();
            RDFNode sourceConstraint = reportModel.asRDFNode(sourceConstraintNode);
            String constraintName = sourceConstraint != null ? sourceConstraint.toString() : "Unknown Constraint";
            String resultMessage = entry.message();
            String uniqueConstraintKey = constraintName + " - " + resultMessage;
            violationCounts.put(uniqueConstraintKey, violationCounts.getOrDefault(uniqueConstraintKey, 0) + 1);
            totalViolations.incrementAndGet();
        });
        if (totalViolations.get() > 0)
            System.out.println("Violation Counts by Individual Constraint:");
        violationCounts.forEach((constraint, count) -> {
            System.out.println("Constraint: " + constraint);
            System.out.println("Violations: " + count);
        });
        System.out.println("Total Number of Violations: " + totalViolations.get());


    }

    private void checkEnableButtons(TextArea mappingContent, TextArea keysContent, TextArea dataContent, Button generateButton, Button validateButton) {
        boolean mappingAndKeysLoaded = !mappingContent.getText().isBlank() && !keysContent.getText().isBlank();
        generateButton.setDisable(!mappingAndKeysLoaded);
        validateButton.setDisable(!mappingAndKeysLoaded || dataContent.getText().isBlank());
    }

    private Map<String, String> parseMappingFile(String mappingContent) {
        Map<String, String> equivalentsMap = new HashMap<>();
        for (String line : mappingContent.split("\\n")) {
            String[] parts = line.split(" equivalent ");
            if (parts.length == 2) {
                String key = parts[0].trim().replace(" ", "_");
                String value = parts[1].trim().replace(".", "").replace(" ", "_");
                equivalentsMap.put(key, value);
            }
        }
        return equivalentsMap;
    }

    private List<List<String>> parseKeysFile(String keysContent, Map<String, String> equivalentsMap) {
        List<List<String>> almostKeys = new ArrayList<>();
        for (String line : keysContent.split("\\n")) {
            if (line.contains("almost keys")) {
                String keysSection = line.split(":")[1].trim();
                String[] keysGroups = keysSection.substring(1, keysSection.length() - 1).split("], \\[");
                for (String group : keysGroups) {
                    List<String> keyList = new ArrayList<>();
                    for (String key : group.replace("[", "").replace("]", "").split(",")) {
                        String trimmedKey = key.trim();
                        keyList.add(equivalentsMap.getOrDefault(trimmedKey, trimmedKey));
                    }
                    almostKeys.add(keyList);
                }
            }
        }
        return almostKeys;
    }
    private void validateAgainstEquivalentData(Map<String, String> equivalentsMap, String dataContent, String shapesContent) {
        validationResultsTextArea.clear();

        if (equivalentsMap.isEmpty() || dataContent.isBlank() || shapesContent.isBlank()) {
            showAlert("Error", "Ensure all files are loaded and SHACL shapes are generated.", Alert.AlertType.ERROR);
            return;
        }

        try {
            for (Map.Entry<String, String> entry : equivalentsMap.entrySet()) {
                dataContent = dataContent.replace(entry.getKey(), entry.getValue());
            }

            Model dataModel = ModelFactory.createDefaultModel();
            Model shapesModel = ModelFactory.createDefaultModel();
            dataModel.read(new StringReader(dataContent), null, "TURTLE");
            shapesModel.read(new StringReader(shapesContent), null, "TURTLE");

            ValidationReport report = ShaclValidator.get().validate(shapesModel.getGraph(), dataModel.getGraph());
            if (report.conforms()) {
                validationResultsTextArea.setText("Validation successful! The data conforms to the SHACL shapes.");
            } else {
                StringWriter writer = new StringWriter();
                report.getModel().write(writer, "TURTLE");
                validationResultsTextArea.setText("Validation failed:\n" + writer);
            }
        } catch (Exception ex) {
            showAlert("Error", "Validation failed: " + ex.getMessage(), Alert.AlertType.ERROR);
        }
    }















    private void generateShapes(List<List<String>> almostKeys) {
        shapesTextArea.clear();
        saveShapesButton.setDisable(true);

        if (almostKeys == null || almostKeys.isEmpty()) {
            showAlert("Error", "No almost keys available to generate shapes.", Alert.AlertType.ERROR);
            return;
        }

        Model model = ModelFactory.createDefaultModel();
        String sh = "http://www.w3.org/ns/shacl#";
        String ex = "http://example.com/ns#";
        String owl = "http://www.w3.org/2002/07/owl#";

        model.setNsPrefix("sh", sh);
        model.setNsPrefix("ex", ex);
        model.setNsPrefix("owl", owl);
        model.setNsPrefix("xsd", XSD.getURI());

        Property rdfType = RDF.type;

        Resource sparqlTarget = model.createResource(sh + "SPARQLTarget");
        Property shTarget = model.createProperty(sh + "target");
        Property shPath = model.createProperty(sh + "path");
        Property shProperty = model.createProperty(sh + "property");
        Property shDatatype = model.createProperty(sh + "datatype");
        Property shMaxCount = model.createProperty(sh + "maxCount");
        Property shSelect = model.createProperty(sh + "select");
        Property shMessage = model.createProperty(sh + "message");
        Property shSeverity = model.createProperty(sh + "severity");
        Property shRule = model.createProperty(sh + "rule");
        Property shSparql = model.createProperty(sh + "sparql");

        Resource shViolation = model.createResource(sh + "Violation");
        Resource tripleRule = model.createResource(sh + "TripleRule");

        int shapeIndex = 1;
        for (List<String> key : almostKeys) {
            Resource shape = model.createResource(ex + "Shape" + shapeIndex++);
            shape.addProperty(rdfType, model.createResource(sh + "NodeShape"));

            // Add property constraints
            for (String property : key) {
                Resource propertyShape = model.createResource();
                propertyShape.addProperty(shPath, model.createProperty(ex + property));
                propertyShape.addProperty(shDatatype, XSD.xstring);
                propertyShape.addProperty(shMaxCount, model.createTypedLiteral(1));
                shape.addProperty(shProperty, propertyShape);
            }

            // Build SPARQL query condition
            String sparqlConditions = buildSparqlConditions(key);
            String sparqlQuery = String.format("""
    PREFIX ex: <http://example.com/ns#>
    SELECT ?this
    WHERE {
        %s
        FILTER (?this != ?other)
    }
    """, sparqlConditions);

            // Add SPARQL constraint
            Resource sparqlConstraint = model.createResource();
            sparqlConstraint.addProperty(rdfType, model.createResource(sh + "SPARQLConstraint"));
            sparqlConstraint.addProperty(shMessage, "Duplicate detected for properties: " + String.join(", ", key));
            sparqlConstraint.addProperty(shSelect, sparqlQuery);
            shape.addProperty(shSparql, sparqlConstraint);

            // Add SPARQL target
            Resource target = model.createResource();
            target.addProperty(rdfType, sparqlTarget);
            target.addProperty(shSelect, sparqlQuery);
            shape.addProperty(shTarget, target);

            // Add severity
            shape.addProperty(shSeverity, shViolation);

            // Consolidate violations for "owl:sameAs"
            String ruleQuery = String.format("""
    PREFIX ex: <http://example.com/ns#>
    PREFIX owl: <http://www.w3.org/2002/07/owl#>
    CONSTRUCT {
        ?this owl:sameAs ?other .
    }
    WHERE {
        %s
        FILTER (?this != ?other)
    }
    """, sparqlConditions);

            Resource rule = model.createResource();
            rule.addProperty(rdfType, tripleRule);
            rule.addProperty(shSelect, ruleQuery);
            shape.addProperty(shRule, rule);

            // Add SPARQL validation constraint to check if owl:sameAs exists
            String validationQuery = String.format("""
    PREFIX ex: <http://example.com/ns#>
    PREFIX owl: <http://www.w3.org/2002/07/owl#>
    SELECT ?this ?other
    WHERE {
        %s
        FILTER (?this != ?other)
        FILTER NOT EXISTS { ?this owl:sameAs ?other }
    }
    """, sparqlConditions);

            Resource validationConstraint = model.createResource();
            validationConstraint.addProperty(rdfType, model.createResource(sh + "SPARQLConstraint"));
            validationConstraint.addProperty(shMessage, "Duplicate with the same properties must be marked as owl:sameAs.");
            validationConstraint.addProperty(shSelect, validationQuery);
            shape.addProperty(shSparql, validationConstraint);
        }

        // Write the generated SHACL shapes to the shapesTextArea
        StringWriter writer = new StringWriter();
        model.write(writer, "TURTLE");
        String generatedShapes = writer.toString();
        javafx.application.Platform.runLater(() -> {
            shapesTextArea.replaceText(generatedShapes);
            saveShapesButton.setDisable(false);
        });
    }

private String buildSparqlConditions(List<String> key) {
    StringBuilder conditions = new StringBuilder();

    // Build the conditions to check the entire key as a combination of properties
    for (int i = 0; i < key.size(); i++) {
        String property = key.get(i);
        conditions.append(String.format("    ?this ex:%s ?%s .\n", property, property));
        conditions.append(String.format("    ?other ex:%s ?%s .\n", property, property));  // Check "other" for duplicates
    }

    return conditions.toString();
}




    private void saveGeneratedShapesToFile() {
        String shapesContent = shapesTextArea.getText();

        FileChooser fileChooser = new FileChooser();
        fileChooser.setTitle("Save SHACL Shapes");
        fileChooser.getExtensionFilters().add(new FileChooser.ExtensionFilter("Turtle Files", "*.ttl"));

            } catch (IOException e) {
                showAlert("Error", "Error saving SHACL Shapes: " + e.getMessage(), Alert.AlertType.ERROR);
            }
        }
    }

    private void openRdfFileChooser() {
        FileChooser fileChooser = new FileChooser();
        fileChooser.getExtensionFilters().add(new FileChooser.ExtensionFilter("RDF Files", "*.nt", "*.ttl"));
        File selectedFile = fileChooser.showOpenDialog(null);
        if (selectedFile != null) {
            selectedFilePath = selectedFile.getAbsolutePath();
            rdfContentTextArea.clear();
            almostKeys = new ArrayList<>();
            outputTextArea.clear();
            saveKeysButton.setDisable(true);
            loadingIndicator.setVisible(true);
            rdfContentTextArea.replaceText("Loading file content... Please wait.");
            new Thread(() -> {
                try {
                    String content = new String(Files.readAllBytes(Paths.get(selectedFile.toURI())));
                    javafx.application.Platform.runLater(() -> {
                        rdfContentTextArea.replaceText(content);
                        loadingIndicator.setVisible(false);
                        generateKeysButton.setDisable(false);
                        generateKeysButton.setOnAction(e -> generateKeys());
                    });
                } catch (IOException ex) {
                    javafx.application.Platform.runLater(() -> {
                        showAlert("Error", "Failed to read the file.", Alert.AlertType.ERROR);
                        loadingIndicator.setVisible(false);
                    });
                }
            }).start();
        }
    }

    private void generateKeys() {
        if (selectedFilePath != null && !selectedFilePath.isEmpty()) {
            String sakeyCommand = "java -jar C:\\Users\\asus\\Downloads\\TER-Materials\\TER-Materials\\sakey.jar \"" + selectedFilePath + "\" 1";
            loadingIndicator.setVisible(true);
            outputTextArea.clear();
            new Thread(() -> {
                try {
                    Process process = new ProcessBuilder(sakeyCommand.split(" ")).start();
                    BufferedReader reader = new BufferedReader(new InputStreamReader(process.getInputStream()));
                    StringBuilder output = new StringBuilder();
                    String line;
                    while ((line = reader.readLine()) != null) {
                        output.append(line).append("\n");
                    }

                    int exitCode = process.waitFor();
                    javafx.application.Platform.runLater(() -> {
                        loadingIndicator.setVisible(false);
                        if (exitCode == 0) {
                            outputTextArea.setText(output.toString());
                            almostKeys = parseAlmostKeysFromOutput(output.toString());
                            saveKeysButton.setDisable(false);
                        } else {
                            showAlert("Error", "The sakey tool encountered an error.", Alert.AlertType.ERROR);
                        }
                    });
                } catch (IOException | InterruptedException ex) {
                    javafx.application.Platform.runLater(() -> {
                        showAlert("Error", "An error occurred while running the sakey tool.", Alert.AlertType.ERROR);
                        loadingIndicator.setVisible(false);
                    });
                }
            }).start();
        } else {
            showAlert("Error", "No RDF file selected.", Alert.AlertType.ERROR);
        }
    }

    private List<List<String>> parseAlmostKeysFromOutput(String output) {
        List<List<String>> keys = new ArrayList<>();
        String almostKeysOutput = output.split("0-almost keys:")[1].split("1-non keys:")[0].trim();
        String[] keyGroups = almostKeysOutput.split("], \\[");
        for (String group : keyGroups) {
            List<String> keyList = Arrays.asList(group.replace("[", "").replace("]", "").split(", "));
            keys.add(keyList);
        }

        return keys;
    }

    private void showAlert(String title, String message, Alert.AlertType alertType) {
        Alert alert = new Alert(alertType);
        alert.setTitle(title);
        alert.setHeaderText(null);
        alert.setContentText(message);
        alert.showAndWait();
    }
